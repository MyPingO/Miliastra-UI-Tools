tructureGiaListValueMessage=function(typeCode,values){
  const type=Number(typeCode),list=values||[];if(!list.length)return new Uint8Array();
  if(type===11)return concatBytes(...list.map(v=>encodeField(1,2,encoder.encode(String(v??'')))));
  if(type===15)return concatBytes(...list.map(v=>encodeField(1,2,concatBytes(encodeField(1,5,floatBytes(v?.x)),encodeField(2,5,floatBytes(v?.y)),encodeField(3,5,floatBytes(v?.z))))));
  if(type===10)return encodeField(1,2,concatBytes(...list.map(v=>floatBytes(v))));
  if(referenceLists.has(type))return concatBytes(...list.map(v=>{const id=BigInt(String(v??'0').trim()||'0');return encodeField(1,2,id===0n?new Uint8Array():encodeField(2,0,id));}));
  if(packedVarintLists.has(type)){
    const packed=concatBytes(...list.map(v=>{if(type===8)return encodeVarint(BigInt.asUintN(64,BigInt(Math.trunc(Number(v)||0))));if(type===9)return encodeVarint(v?1:0);return encodeVarint(BigInt(String(v??'0').trim()||'0'));}));
    return encodeField(1,2,packed);
  }
  return new Uint8Array();
};

function readListEditor(item){
  const info=listInfo(item.typeCode);if(!info)return true;const output=clone(item.listValues||[]);
  for(const c of elements.structureListEditor.querySelectorAll('[data-structure-list-index]')){
    const i=Number(c.dataset.structureListIndex);
    if(info.elementKind==='vector3'){output[i]??={x:0,y:0,z:0};const n=Number(c.value);if(!Number.isFinite(n)){alert(`Vector3 list item ${i+1} must be numeric.`);return false;}output[i][c.dataset.structureListAxis]=n;}
    else if(info.elementKind==='string')output[i]=c.value;
    else if(info.elementKind==='integer'){const n=Number(c.value);if(!Number.isFinite(n)){alert(`Integer list item ${i+1} must be numeric.`);return false;}output[i]=Math.trunc(n);}
    else if(info.elementKind==='bool')output[i]=c.value==='true';
    else if(info.elementKind==='float'){const n=Number(c.value);if(!Number.isFinite(n)){alert(`Float list item ${i+1} must be numeric.`);return false;}output[i]=n;}
    else if(info.elementKind==='id'){try{const n=BigInt(c.value.trim()||'0');if(n<0n||n>0xFFFFFFFFFFFFFFFFn)throw 0;output[i]=n.toString();}catch{alert(`${info.label} list item ${i+1} must be an unsigned integer ID.`);return false;}}
  }
  item.listValues=output;item.defaultValue=structureListDefaultSummary(item.typeCode,output);return true;
}
